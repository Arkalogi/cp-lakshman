            const qty = Math.floor(Math.random() * 2000 + 1);

            const div = document.createElement('div');
            div.className = 'p-3 hover:bg-surface-container-low flex justify-between gap-4 transition-all duration-300 opacity-0 -translate-y-2';

                <div class="flex gap-2">
                    <span class="text-outline">${timestamp}</span>
                    <span class="${isBuy ? 'text-success' : 'text-danger'} font-bold">${isBuy ? 'BUY' : 'SELL'}</span>
                    <span class="text-on-surface">${stock}</span>
                </div>
                <div class="text-right">
                    <span class="text-on-surface-variant">${price}</span>
                    <span class="text-outline ml-2">x${qty}</span>
                </div>
            `;
            
            feed.prepend(div);
            
            // Trigger animation
            requestAnimationFrame(() => {
                div.classList.remove('opacity-0', '-translate-y-2');
